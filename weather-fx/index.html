Cómo se conecta a JA-VTT 
const fx = new WeatherFX({
  container: document.querySelector('#tablero'),
  source: miCanvas2D        // tu tablero actual
});

fx.use('rain');
fx.set({ intensity: .8 });

// cuando repintes el mapa:
fx.invalidateSource();

// una gota puntual (conjuro, paso, proyectil):
fx.drop(x, y, { radius: 26, strength: 1.2 });

// fichas que se mojan y se refractan:
fx.world.addChild(spriteDeLaFicha);
